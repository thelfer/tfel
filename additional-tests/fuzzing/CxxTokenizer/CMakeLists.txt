# libFuzzer based analyses of the tfel::utilities::CxxTokenizer class.
#
# Those analyses are only configured when the 'enable-libfuzzer-tests' option
# is enabled. As opposed to the other tests of TFEL, they are neither built nor
# run by 'make check': the duration of a fuzzing analysis is a priori unlimited,
# the fuzzing executables are excluded from the default target and no test is
# registered with add_test. Two targets are dedicated to those analyses:
#
# - build-cxxtokenzier-libfuzzer-tests: build the fuzzing executables,
# - cxxtokenzier-libfuzzer-tests: build the fuzzing executables and run the
#   analyses.
#
# The duration of the analyses and the size of the inputs they generate are
# controlled by the 'libfuzzer-tests-max-total-time' and
# 'libfuzzer-tests-max-len' options. The database of an analysis is given by the
# '<name>-database' option, where <name> is the name of the analysis.

# libFuzzer is a sanitizer of clang, provided by compiler-rt.
if(NOT ((CMAKE_CXX_COMPILER_ID STREQUAL "Clang") OR
        (CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang")))
  message(FATAL_ERROR
    "the libFuzzer tests can only be built with clang: "
    "${CMAKE_CXX_COMPILER_ID} (${CMAKE_CXX_COMPILER_VERSION}) is not "
    "supported")
endif()

# The libFuzzer runtime is not always installed along clang since it is
# provided by the compiler-rt package, which is a separate package on some
# distributions. The check links the fuzzer runtime, which is not only a test
# for the availability of the compiler flags.
set(safe_tfel_cmake_required_flags "${TFEL_CMAKE_REQUIRED_FLAGS}")
set(TFEL_CMAKE_REQUIRED_FLAGS "-fsanitize=fuzzer,address ")
TFEL_CHECK_CXX_SOURCE_COMPILES(
  "#include <cstddef>
   #include <cstdint>
   extern \"C\" int
   LLVMFuzzerTestOneInput(const uint8_t* const data, const size_t size) {
     return (size > 0) ? 0 : 1;
   }"
  TFEL_HAVE_LIBFUZZER)
set(TFEL_CMAKE_REQUIRED_FLAGS "${safe_tfel_cmake_required_flags}")
if(NOT TFEL_HAVE_LIBFUZZER)
  message(FATAL_ERROR
    "the libFuzzer runtime is not available with "
    "${CMAKE_CXX_COMPILER_ID} (${CMAKE_CXX_COMPILER_VERSION}): "
    "the compiler-rt package may be missing")
endif()

set(libfuzzer-tests-max-total-time "60" CACHE STRING
  "duration of each libFuzzer analysis, in seconds")
set(libfuzzer-tests-max-len "65536" CACHE STRING
  "maximum length of the inputs generated by libFuzzer, in bytes (0: no limit)")

# target building the fuzzing executables. It is created before the fuzzing
# tests are declared since those tests are added as dependencies of this
# target.
add_custom_target(build-cxxtokenzier-libfuzzer-tests)

# The sources of the tfel::utilities::CxxTokenizer class are compiled with
# the fuzzing executables instead of being taken from the TFELUtilities
# library: the sanitizers are only effective if the analysed code is
# instrumented, which is not the case of the libraries of the project since
# adding the sanitizers to those libraries would affect all the targets of the
# build. Any source file the tokenizer depends on shall be added to this list.
set(tfel_cxxtokenizer_sources
  ${PROJECT_SOURCE_DIR}/src/Utilities/CxxTokenizer.cxx
  ${PROJECT_SOURCE_DIR}/src/Utilities/CxxTokenizerOptions.cxx
  ${PROJECT_SOURCE_DIR}/src/Utilities/CxxKeywords.cxx
  ${PROJECT_SOURCE_DIR}/src/Utilities/StringAlgorithms.cxx
  ${PROJECT_SOURCE_DIR}/src/Utilities/Token.cxx)

# tfel_add_cxxtokenizer_libfuzzer_test(<name> <source> [<database>...])
#
# add a libFuzzer based analysis of the tfel::utilities::CxxTokenizer class,
# where:
# - <name>     : name of the analysis, which is also the name of the fuzzing
#                executable,
# - <source>   : source file defining the LLVMFuzzerTestOneInput function,
# - <database> : files or directories used as database, that is as initial
#                corpus of the analysis. A directory is meant recursively and
#                files sharing the same name overwrite each other.
#
# If the first element of the database is 'FLATTEN_NAMES', the files of the
# database are copied to the corpus with a name built from their path instead of
# their only name, which is needed when the database holds files sharing the
# same name, as the sources of a project spread over several directories do.
#
# The corpus of an analysis is stored in the build directory and fed with the
# database at configure time: it gathers both the inputs of the database and
# the inputs of interest found by the previous analyses, so that an analysis
# is resumed when it is run again. The inputs leading to a failure are kept in
# a separate directory.
function(tfel_add_cxxtokenizer_libfuzzer_test name source)
  if(NOT ARGN)
    message(FATAL_ERROR "tfel_add_cxxtokenizer_libfuzzer_test: "
      "no database is given for the '${name}' analysis")
  endif()
  set(database ${ARGN})
  set(flatten_names FALSE)
  list(GET database 0 first_entry)
  if(first_entry STREQUAL "FLATTEN_NAMES")
    set(flatten_names TRUE)
    list(REMOVE_AT database 0)
    if(NOT database)
      message(FATAL_ERROR "tfel_add_cxxtokenizer_libfuzzer_test: "
        "no database is given for the '${name}' analysis")
    endif()
  endif()
  # the executable is excluded from the default target and is not added to the
  # build-tests target: the fuzzing analyses are neither built nor run by
  # 'make check'
  add_executable(${name} EXCLUDE_FROM_ALL ${source} ${tfel_cxxtokenizer_sources})
  target_include_directories(${name} PRIVATE ${PROJECT_SOURCE_DIR}/include)
  target_compile_options(${name} PRIVATE
    -fsanitize=fuzzer-no-link -fsanitize=address
    -fno-omit-frame-pointer -g)
  target_link_options(${name} PRIVATE -fsanitize=fuzzer -fsanitize=address)
  # the address sanitizer maps its allocator at a fixed address of the address
  # space, which can be the address the dynamic loader chooses for a position
  # independent executable. Such a collision, which depends on how the address
  # space is laid out, makes the fuzzing executable crash when it starts.
  # Fuzzing executables are analysis tools, which are not installed, so that
  # position independent code is not needed.
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_options(${name} PRIVATE -fno-pie)
    target_link_options(${name} PRIVATE -no-pie)
  endif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_dependencies(build-cxxtokenzier-libfuzzer-tests ${name})
  set(corpus ${CMAKE_CURRENT_BINARY_DIR}/${name}-corpus)
  file(MAKE_DIRECTORY ${corpus} ${CMAKE_CURRENT_BINARY_DIR}/${name}-artifacts)
  foreach(entry ${database})
    if(IS_DIRECTORY ${entry})
      file(GLOB_RECURSE entries ${entry}/*)
    else()
      set(entries ${entry})
    endif()
    foreach(f ${entries})
      if(NOT IS_DIRECTORY ${f})
        if(flatten_names)
          # the name of the copy in the corpus is built from the path of the
          # file, so that two files of the database sharing the same name do not
          # overwrite each other and so that the origin of every input is kept
          file(RELATIVE_PATH r ${PROJECT_SOURCE_DIR} ${f})
          string(REPLACE "/" "_" corpus_entry ${r})
          configure_file(${f} ${corpus}/${corpus_entry} COPYONLY)
        else()
          file(COPY ${f} DESTINATION ${corpus})
        endif()
      endif()
    endforeach()
  endforeach()
  list(APPEND tfel_cxxtokenizer_libfuzzer_tests ${name})
  set(tfel_cxxtokenizer_libfuzzer_tests ${tfel_cxxtokenizer_libfuzzer_tests}
    PARENT_SCOPE)
endfunction(tfel_add_cxxtokenizer_libfuzzer_test name source)

# Database of the 'mfront-fuzzing' analysis: the contents of mfront files,
# which the analysis tokenizes with the options mfront uses.
#
# The aim of this analysis is to use all the mfront files of
# mfront/tests/behaviours as database. Setting it up and checking it is easier
# with a single file, which is why the 'mfront-fuzzing-database' option
# defaults to 'single-file', which uses the Norton.mfront file, whereas 'all'
# uses all those files.
set(mfront-fuzzing-database "single-file" CACHE STRING
  "database of the 'mfront-fuzzing' analysis: 'single-file' or 'all'")
set_property(CACHE mfront-fuzzing-database PROPERTY STRINGS single-file all)
if(mfront-fuzzing-database STREQUAL "single-file")
  set(mfront_fuzzing_database
    ${PROJECT_SOURCE_DIR}/mfront/tests/behaviours/Norton.mfront)
elseif(mfront-fuzzing-database STREQUAL "all")
  file(GLOB_RECURSE mfront_fuzzing_database
    ${PROJECT_SOURCE_DIR}/mfront/tests/behaviours/*.mfront)
else()
  message(FATAL_ERROR
    "invalid value '${mfront-fuzzing-database}' for the "
    "'mfront-fuzzing-database' option: expected 'single-file' or 'all'")
endif()

tfel_add_cxxtokenizer_libfuzzer_test(
  mfront-fuzzing mfront-fuzzing.cxx ${mfront_fuzzing_database})

# Database of the 'cxx-fuzzing' analysis: the `C++` files of the project, which
# the analysis tokenizes with the default options of the tokenizer, that is as
# `C++` files, as `tfel-doc` and `tfel-check` do.
#
# The aim of this analysis is to use all the `C++` files of the project as
# database, which is four times the database of the 'mfront-fuzzing' analysis
# and a much more various one: setting it up and checking it is easier with a
# single file, which is why the 'cxx-fuzzing-database' option defaults to
# 'single-file', which uses the main source file of the tokenizer, whereas 'all'
# uses all the `C++` files of the project.
set(cxx-fuzzing-database "single-file" CACHE STRING
  "database of the 'cxx-fuzzing' analysis: 'single-file' or 'all'")
set_property(CACHE cxx-fuzzing-database PROPERTY STRINGS single-file all)
if(cxx-fuzzing-database STREQUAL "single-file")
  set(cxx_fuzzing_database
    ${PROJECT_SOURCE_DIR}/src/Utilities/CxxTokenizer.cxx)
elseif(cxx-fuzzing-database STREQUAL "all")
  file(GLOB_RECURSE cxx_fuzzing_database
    ${PROJECT_SOURCE_DIR}/*.c ${PROJECT_SOURCE_DIR}/*.cc
    ${PROJECT_SOURCE_DIR}/*.h ${PROJECT_SOURCE_DIR}/*.hh
    ${PROJECT_SOURCE_DIR}/*.cpp ${PROJECT_SOURCE_DIR}/*.hpp
    ${PROJECT_SOURCE_DIR}/*.cxx ${PROJECT_SOURCE_DIR}/*.hxx
    ${PROJECT_SOURCE_DIR}/*.ixx)
  # the sources generated by a build of the project are not a part of the
  # database: an in tree build would otherwise add the sources generated by
  # mfront and by cmake itself to the corpus of the analysis
  set(tfel_fuzzing_sources)
  foreach(f ${cxx_fuzzing_database})
    file(RELATIVE_PATH r ${CMAKE_BINARY_DIR} ${f})
    if(r MATCHES "^\\.\\.")
      list(APPEND tfel_fuzzing_sources ${f})
    endif()
  endforeach()
  set(cxx_fuzzing_database ${tfel_fuzzing_sources})
  list(REMOVE_DUPLICATES cxx_fuzzing_database)
  # the five files below are rejected by the tokenizer, so that they cannot be a
  # part of the corpus of the analysis: libFuzzer would report the exception
  # thrown for such an input as soon as it replays it, whereas an exception is
  # the expected behaviour for the invalid inputs the fuzzer generates.
  #
  # The first four files are the inputs of the tools of the project, whose names
  # kept the extension of the sources they were made from whereas they are not
  # `C++` files:
  # - an mfront file whose '@Description' section holds '§8.4.2 Page 34', which
  #   is not a valid `C++` token,
  # - two inputs of the `Pleiades` code, whose contents hold multi characters
  #   literals, as `'IsotropicUmatBehaviour'`,
  # - an input of tfel-check, whose contents hold multi characters literals, as
  #   `'../../..//CELAENO/src/celaenoMesh --scheme=Unicell2'`.
  set(tfel_fuzzing_rejected_files
    ${PROJECT_SOURCE_DIR}/docs/mfront/mfront/SICCreepBehaviour.cxx
    ${PROJECT_SOURCE_DIR}/docs/mfront/mfront/SICCreepBehaviour-ple.cxx
    ${PROJECT_SOURCE_DIR}/docs/mfront/mfront/test-T.cxx
    ${PROJECT_SOURCE_DIR}/docs/tfel-check/img/celaeno.cpp
    # this last file is a source which is not built by the project: it is an old
    # copy of 'PleiadesMaterialPropertyInterface-2.0.cxx', which includes a
    # header that does not exist anymore and holds an unterminated string
    # literal, so that it is not a valid `C++` file
    ${PROJECT_SOURCE_DIR}/mfront/tests/properties/interfaces/src/MFrontPleiadesMaterialPropertyInterface-2.0.cxx)
  list(REMOVE_ITEM cxx_fuzzing_database ${tfel_fuzzing_rejected_files})
else()
  message(FATAL_ERROR
    "invalid value '${cxx-fuzzing-database}' for the "
    "'cxx-fuzzing-database' option: expected 'single-file' or 'all'")
endif()

# the database of the 'cxx-fuzzing' analysis gathers the sources of the whole
# project, spread over many directories, and 112 of their names are duplicated,
# so that the names of the copies in the corpus are built from their paths to
# keep the origin of every input
tfel_add_cxxtokenizer_libfuzzer_test(
  cxx-fuzzing cxx-fuzzing.cxx FLATTEN_NAMES ${cxx_fuzzing_database})

# Running the analyses. The corpus directory is the only directory given to
# libFuzzer, so that the inputs of interest are written in the build directory
# and never in the source directory.
set(tfel_cxxtokenizer_libfuzzer_commands
  COMMAND ${CMAKE_COMMAND} -E echo
  "running the libFuzzer analyses of the CxxTokenizer class")
foreach(name ${tfel_cxxtokenizer_libfuzzer_tests})
  list(APPEND tfel_cxxtokenizer_libfuzzer_commands
    COMMAND ${CMAKE_COMMAND} -E echo "analysis '${name}'"
    COMMAND $<TARGET_FILE:${name}>
    "-artifact_prefix=${CMAKE_CURRENT_BINARY_DIR}/${name}-artifacts/"
    "-max_len=${libfuzzer-tests-max-len}"
    "-max_total_time=${libfuzzer-tests-max-total-time}"
    "${CMAKE_CURRENT_BINARY_DIR}/${name}-corpus")
endforeach()

add_custom_target(cxxtokenzier-libfuzzer-tests
  ${tfel_cxxtokenizer_libfuzzer_commands}
  DEPENDS build-cxxtokenzier-libfuzzer-tests
  USES_TERMINAL)
